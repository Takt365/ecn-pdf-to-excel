*&---------------------------------------------------------------------*
*&      Form  F_READ_CHECK
*&---------------------------------------------------------------------*
*       データ存在チェック:メイン
*----------------------------------------------------------------------*
FORM F_READ_CHECK.
  DATA: L_SUBRC LIKE SY-SUBRC.         "リターンコード

** 逆展開
  PERFORM F_CALL_FUNC   USING P_MATNR
                     CHANGING L_SUBRC.

** 対象なしのケース
  IF L_SUBRC <> 0.
    MESSAGE E005(ZB) WITH C_MSG11.
  ENDIF.
ENDFORM.                               " F_READ_CHECK

*&---------------------------------------------------------------------*
*&      Form  F_CALL_FUNC
*&---------------------------------------------------------------------*
*       汎用モジュール ：逆展開
*----------------------------------------------------------------------*
*      -->P_MATNR    品目コード
*      <--P_BISMT    リターンコード
*----------------------------------------------------------------------*
FORM F_CALL_FUNC   USING P_MATNR
                CHANGING P_SUBRC.
  CLEAR:
    TAB_WRK1.                                               "展開用 1
  REFRESH:
    TAB_WRK1.                                               "展開用 1

** 逆展開
  CALL FUNCTION 'CS_WHERE_USED_MAT'
    EXPORTING
      DATUB                      = P_DATUB
      DATUV                      = P_DATUV
      MATNR                      = P_MATNR
      POSTP                      = ' '
      STLAN                      = P_STLAN
      WERKS                      = P_WERKS
    TABLES
      WULTB                      = TAB_WRK1
      EQUICAT                    = TAB_DMY1
      KNDCAT                     = TAB_DMY2
      MATCAT                     = TAB_DMY3
      STDCAT                     = TAB_DMY4
      TPLCAT                     = TAB_DMY5
    EXCEPTIONS
      CALL_INVALID               = 1
      MATERIAL_NOT_FOUND         = 2
      NO_WHERE_USED_REC_FOUND    = 3
      NO_WHERE_USED_REC_SELECTED = 4
      NO_WHERE_USED_REC_VALID    = 5
      OTHERS                     = 6.

  P_SUBRC = SY-SUBRC.
ENDFORM.                               " F_CALL_FUNC

*&---------------------------------------------------------------------*
*&      Form  F_GEN_TABLE
*&---------------------------------------------------------------------*
*       データ抽出：展開＆テーブル作成
*----------------------------------------------------------------------*
FORM F_GEN_TABLE.
  DATA: L_CNT_TABLE TYPE I,            "テーブル件数
        L_SUBRC LIKE SY-SUBRC.         "リターンコード

** 合計レコード区分=*を削除            2006/04/17 add
  DELETE TAB_WRK1 WHERE SUMFG = '*'.

** 初回
  LOOP AT TAB_WRK1.
    CHECK TAB_WRK1-LOEKZ = ' '.        "対象判定
    TAB_WRK21-INTRM = TAB_WRK1-MATNR.
    TAB_WRK21-MATNR = TAB_WRK1-MATNR.
    TAB_WRK21-IDNRK = TAB_WRK1-IDNRK.
    TAB_WRK21-MENGE = TAB_WRK1-MENGE.
    TAB_WRK21-MEINS = TAB_WRK1-MEINS.
    TAB_WRK21-AENNR = TAB_WRK1-AENNR.
    TAB_WRK21-DATUV = TAB_WRK1-DATUV.
    APPEND TAB_WRK21.
    CLEAR  TAB_WRK21.
  ENDLOOP.

** 展開用テーブル2-1が0件になるまでループ
  DESCRIBE TABLE TAB_WRK1 LINES L_CNT_TABLE.

  WHILE L_CNT_TABLE <> 0.
    LOOP AT TAB_WRK21.
      PERFORM F_CALL_FUNC   USING TAB_WRK21-MATNR  "逆展開
                         CHANGING L_SUBRC.
      CASE L_SUBRC.
        WHEN 0.                        "上位レベルあり
          PERFORM F_GEN_WRK22.
*        WHEN 3 OR 4.                   "最上位レベル
        WHEN 2 OR 3 OR 4 OR 5.         "最上位レベル 01/01/30
          PERFORM F_GEN_PRT.
        WHEN OTHERS.
          MESSAGE E006(ZB) WITH 'CS_WHERE_USED_MAT'.
      ENDCASE.
    ENDLOOP.
    CLEAR   TAB_WRK21.
    REFRESH TAB_WRK21.
    TAB_WRK21[] = TAB_WRK22[].
    DESCRIBE TABLE TAB_WRK21 LINES L_CNT_TABLE.
    CLEAR   TAB_WRK22.
    REFRESH TAB_WRK22.
  ENDWHILE.

ENDFORM.                               " F_GEN_TABLE

*&---------------------------------------------------------------------*
*&      Form  F_GEN_WRK22
*&---------------------------------------------------------------------*
*       データ抽出：次の上位レベルのテーブルを作成
*----------------------------------------------------------------------*
FORM F_GEN_WRK22.
  LOOP AT TAB_WRK1.
    CHECK TAB_WRK1-LOEKZ = ' '.        "対象判定
    TAB_WRK22-INTRM = TAB_WRK21-INTRM.
    TAB_WRK22-MATNR = TAB_WRK1-MATNR.
    TAB_WRK22-IDNRK = TAB_WRK21-IDNRK.
    TAB_WRK22-MENGE = TAB_WRK1-MENGE * TAB_WRK21-MENGE.
    TAB_WRK22-MEINS = TAB_WRK21-MEINS.
    TAB_WRK22-AENNR = TAB_WRK21-AENNR.
    TAB_WRK22-DATUV = TAB_WRK21-DATUV.
    APPEND TAB_WRK22.
    CLEAR  TAB_WRK22.
  ENDLOOP.

ENDFORM.                               " F_GEN_WRK22

*&---------------------------------------------------------------------*
*&      Form  F_GEN_PRT
*&---------------------------------------------------------------------*
*       データ抽出：最上位レベルのテーブルを作成
*----------------------------------------------------------------------*
FORM F_GEN_PRT.
  DATA: L_BISMT LIKE ENT1034-BISMT,    "機種コード
        L_MATKL LIKE ENT1034-MATKL.    "品目グループ

  TAB_PRT-INTRM = TAB_WRK21-INTRM.
  TAB_PRT-MATNR = TAB_WRK21-MATNR.
  TAB_PRT-IDNRK = TAB_WRK21-IDNRK.
  TAB_PRT-MENGE = TAB_WRK21-MENGE.
  TAB_PRT-MEINS = TAB_WRK21-MEINS.
  TAB_PRT-AENNR = TAB_WRK21-AENNR.
  TAB_PRT-DATUV = TAB_WRK21-DATUV.

** 品目テキスト　取得 (親)
  PERFORM F_READ_ENT1034 USING TAB_PRT-MATNR
                      CHANGING TAB_PRT-MAKTX TAB_PRT-BISMT L_MATKL.
** 品目テキスト　取得 (直上)
  PERFORM F_READ_ENT1034 USING TAB_PRT-INTRM
                      CHANGING TAB_PRT-INTTX L_BISMT L_MATKL.
  APPEND TAB_PRT.
  CLEAR  TAB_PRT.

ENDFORM.                               " F_GEN_PRT

*&---------------------------------------------------------------------*
*&      Form  F_READ_ENT1034
*&---------------------------------------------------------------------*
*       品目テキスト、他　取得
*----------------------------------------------------------------------*
*      -->P_MATNR    品目コード
*      <--P_MAKTX    品目テキスト
*      <--P_BISMT    機種コード
*      <--P_MATKL    品目グループ
*----------------------------------------------------------------------*
FORM F_READ_ENT1034 USING P_MATNR
                 CHANGING P_MAKTX P_BISMT P_MATKL.
  CLEAR: ENT1034-MAKTX,
         ENT1034-BISMT,
         ENT1034-MATKL.

  SELECT SINGLE
      MATKL                            "品目グループ
      MAKTX                            "品目テキスト
      BISMT                            "機種コード
      FROM ENT1034
    INTO (ENT1034-MATKL, ENT1034-MAKTX, ENT1034-BISMT)
    WHERE  MATNR = P_MATNR
    AND    SPRAS = 'JA'.

  P_MAKTX = ENT1034-MAKTX.
  P_BISMT = ENT1034-BISMT.
  P_MATKL = ENT1034-MATKL.
ENDFORM.                               " F_READ_ENT1034

*&---------------------------------------------------------------------*
*&      Form  F_WRITE_REPORT
*&---------------------------------------------------------------------*
*       レポート出力
*----------------------------------------------------------------------*
FORM F_WRITE_REPORT.
  DATA: L_BISMT LIKE ENT1034-BISMT.    "機種コード

** 品目テキスト,他　取得(ヘッダー)
  PERFORM F_READ_ENT1034   USING P_MATNR
                        CHANGING G_MAKTX L_BISMT G_MATKL.

  LOOP AT TAB_PRT.
    MOVE-CORRESPONDING TAB_PRT TO REC_LIST.
** 親品目ブレイク
    AT NEW MATNR.
      REC_SAVE-BISMT = REC_LIST-BISMT.
      REC_SAVE-MATNR = REC_LIST-MATNR.
      REC_SAVE-MAKTX = REC_LIST-MAKTX.
    ENDAT.
** 明細出力
    PERFORM F_WRITE_DETAIL.
  ENDLOOP.

ENDFORM.                               " F_WRITE_REPORT

*--INS END  - T5756 - 20150129 --*
  WRITE: /002 L_BISMT,                 "機種コード
          016 L_MATNR,                 "親品目コード
*--UPD START- T5756 - 20150129 --*
*          030 REC_SAVE-MAKTX+0(25),    "親品目テキスト
          030 L_MAKTX,                 "親品目テキスト
*--UPD END  - T5756 - 20150129 --*
          057 L_INTRM,                 "直上品目コード
*--UPD START- T5756 - 20150129 --*
*          073 TAB_PRT-INTTX+0(25),     "直上品目テキスト
          073 L_INTTX,                 "直上品目テキスト
*--UPD END  - T5756 - 20150129 --*
          098 TAB_PRT-MENGE,           "数量
          117 TAB_PRT-MEINS,           "単位
          122 TAB_PRT-AENNR+0(06),     "設変番号
          131 TAB_PRT-DATUV.           "有効開始日

  CLEAR: REC_SAVE.
ENDFORM.                               " F_WRITE_DETAIL
