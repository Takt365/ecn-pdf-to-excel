*&---------------------------------------------------------------------*
*&      Form  F_READ_CHECK
*&---------------------------------------------------------------------*
*       データ存在チェック:メイン
*----------------------------------------------------------------------*
FORM F_READ_CHECK.
** 変更マスタ
  SELECT SINGLE *
    FROM AENR
    WHERE AENNR = P_AENNR.

** 対象なしのケース
  IF SY-SUBRC <> 0.
    MESSAGE E005(ZB).
  ENDIF.
ENDFORM.                               " F_READ_CHECK

          013 SY-CPROG,                "プログラムID
          158 'PAGE : ',

          016 G_KISYU1,

*--UPD START- T5756 - 20150123 --*

*          108 REC_WRK1-AETXT.

*----------------------------------------------------------------------*
FORM F_READ_AUSP.
  DATA: L_FLD(20)  TYPE C,             "フィールド　シンボル用
        L_IDX(02)  TYPE N.             "インデックス
  FIELD-SYMBOLS:
        <F1>       TYPE ANY.

** 内部特性　セット
  DO C_ATNAM_MAX TIMES.
    L_IDX = SY-INDEX.
    CONCATENATE 'C_ATNAM' L_IDX INTO L_FLD.
    ASSIGN (L_FLD) TO <F1>.            "C_ATNAMXX : ZPP_XX
    PERFORM F_READ_CABN USING <F1> CHANGING S_ATINN-LOW.

    S_ATINN-SIGN   = 'I'.
    S_ATINN-OPTION = 'EQ'.
    APPEND S_ATINN.
    CLEAR  S_ATINN.
  ENDDO.
** 内部特性　取得
  CLEAR: AUSP-ATINN,
         AUSP-ATWRT,
         AUSP-ATFLV.
  SELECT
      ATINN                            "内部特性
      ATWRT                            "特性値
      ATFLV                            "開始値
      FROM AUSP
  INTO CORRESPONDING FIELDS OF TABLE TAB_AUSP
    WHERE  OBJEK = P_AENNR
    AND    ATINN IN S_ATINN
    AND    KLART = '025'.
** CAWNT:特性値取得
  LOOP AT TAB_AUSP.
    LOOP AT S_ATINN WHERE LOW = TAB_AUSP-ATINN.
      L_IDX = SY-TABIX.
    ENDLOOP.
** (対象INDEX : 04, 07 - 19)
    CHECK L_IDX = 04 OR ( L_IDX >= 07 AND L_IDX <= 19 ).
** (特性値)
    CLEAR: CAWN-ATZHL,
           CAWN-ADZHL.
    SELECT
        ATZHL                          "内部カウンタ
        ADZHL                          "アーカイブ用内部カウンタ
        FROM CAWN UP TO 1 ROWS
      INTO (CAWN-ATZHL, CAWN-ADZHL)
      WHERE  ATINN = TAB_AUSP-ATINN
      AND    ATWRT = TAB_AUSP-ATWRT.
    ENDSELECT.
    CHECK SY-SUBRC = 0.
** (特性値テキスト)
    CLEAR: CAWNT-ATWTB.
    SELECT SINGLE
        ATWTB                          "特性値
        FROM CAWNT
      INTO (CAWNT-ATWTB)
      WHERE  ATINN = TAB_AUSP-ATINN
      AND    ATZHL = CAWN-ATZHL
      AND    SPRAS = 'JA'
      AND    ADZHL = CAWN-ADZHL.
    CHECK SY-SUBRC = 0.
** (テーブル更新)
    TAB_AUSP-ATWTB = CAWNT-ATWTB.
    MODIFY TAB_AUSP.
  ENDLOOP.
ENDFORM.                               " F_READ_AUSP

*&      Form  F_READ_DATA2
*&---------------------------------------------------------------------*
*       データ抽出：詳細用
*----------------------------------------------------------------------*
FORM F_READ_DATA2.
  DATA:  L_CNT_ADD    TYPE I,          "件数カウント: 追加
         L_CNT_DEL    TYPE I,          "件数カウント: 削除
         L_FLG_STLKN(01)  TYPE C,      "明細ノード番号 フラグ
         L_STLKN      LIKE STAS-STLKN, "明細ノード番号 ワーク
         L_SAVE_STLKN LIKE STAS-STLKN, "明細ノード番号 セーブ
         L_STR_ADD    LIKE STPO-STLKN, "追加 - 最初のデータ
         L_STR_DEL    LIKE STPO-STLKN, "削除 - 最初のデータ
         L_END_ADD    LIKE STPO-STLKN, "追加 - 最後のデータ
         L_END_DEL    LIKE STPO-STLKN. "削除 - 最後のデータ

** BOM明細選択
  SELECT
      STLTY STLNR STLKN              "BOMカテゴリ 部品表 明細ノード番号
      LKENZ STVKN                    "削除フラグ BOM明細の継承ノード番号
      FROM STAS
    INTO CORRESPONDING FIELDS OF TABLE TAB_WRK3
    WHERE AENNR = P_AENNR
    AND   STLTY = 'M'
    ORDER BY STLTY STLNR STVKN STLKN ASCENDING.

** 詳細用ワーク作成

  LOOP AT TAB_WRK3.
    L_STLKN = TAB_WRK3-STLKN.
** (親品目　取得)
    AT NEW STLNR.
      CLEAR: MAST-MATNR.
      SELECT SINGLE
          MATNR                        "品目コード
          FROM MAST
        INTO (MAST-MATNR)
        WHERE STLNR = TAB_WRK3-STLNR.
*      PERFORM F_READ_MAKT USING TAB_WRK5-MATNR  CHANGING G_MAKTX. 20170525 Shin
      PERFORM F_READ_MAKT USING MAST-MATNR  CHANGING G_MAKTX.
    ENDAT.
** (クリア)
    AT NEW STVKN.
      CLEAR: L_CNT_ADD, L_CNT_DEL, L_FLG_STLKN,
             L_STR_ADD, L_STR_DEL, L_END_ADD, L_END_DEL.
      L_SAVE_STLKN = L_STLKN.
    ENDAT.
** (データセーブと件数カウント)
    IF TAB_WRK3-STLKN <> L_SAVE_STLKN.
      L_FLG_STLKN = 'Y'.
    ENDIF.
    IF TAB_WRK3-LKENZ <> ' ' AND L_STR_DEL = 0.
      L_STR_DEL = TAB_WRK3-STLKN.      "初回　変更・削除用
    ENDIF.
    IF TAB_WRK3-LKENZ =  ' ' AND L_STR_ADD = 0.
      L_STR_ADD = TAB_WRK3-STLKN.      "初回　追加用
    ENDIF.
    IF TAB_WRK3-LKENZ = ' '.
      L_CNT_ADD = L_CNT_ADD + 1.
      L_END_ADD = TAB_WRK3-STLKN.      "最後　変更・追加用
    ELSE.
      L_CNT_DEL = L_CNT_DEL + 1.
      L_END_DEL = TAB_WRK3-STLKN.      "最後　削除用
    ENDIF.
** (テーブル：詳細用ワーク作成)
    AT END OF STVKN.
      CHECK L_CNT_ADD > 0 OR L_CNT_DEL > 0.
** (追加・削除・変更の判定とBOM明細・副明細 取得)
      IF L_CNT_ADD = L_CNT_DEL.        "変更のケース
        CHECK L_FLG_STLKN = 'Y'.       "明細ノード番号同一は対象外
        PERFORM F_GET_BOM  USING 'X' L_STR_DEL.
        PERFORM F_GET_BOM  USING ' ' L_END_ADD.
      ELSE.
        IF L_CNT_ADD > L_CNT_DEL.      "追加のケース
          PERFORM F_GET_BOM  USING ' ' L_STR_ADD.
        ELSE.                          "削除のケース
          PERFORM F_GET_BOM  USING 'X' L_END_DEL.
        ENDIF.
      ENDIF.
    ENDAT.
  ENDLOOP.
ENDFORM.                               " F_READ_DATA2
